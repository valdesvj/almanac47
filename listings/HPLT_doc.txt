; ======================================================================
; HPLT - Hc/Zn dot plot with the statistics plot (PLSTAT)
; IN : Z = JD, Y = lat, X = lon
; Clears stats, Sigma+ (x = Zn, y = Hc) for Sun and stars above horizon
; ======================================================================
;
; ---- store inputs, clear statistics, Sun
0001  LBL "HPLT"
0002  STO 92
0003  R↓
0004  STO 91
0005  R↓
0006  STO 90
0007  XEQ "HCZI"
0008  CLΣ
0009  RCL 90
0010  XEQ "SUNA"
0011  RCL 77
0012  RCL 81
0013  XEQ 52
0014  RCL 96
0015  X>0?
0016  XEQ 50
0017  1.058
0018  STO 93
;
; ---- star loop 1-58
0019  LBL 51
0020  RCL 93
0021  IP
0022  STO 82
0023  XEQ "STR2"
0024  XEQ 52
0025  RCL 96
0026  X>0?
0027  XEQ 50
0028  ISG 93
0029  GTO 51
; ---- show plot
0030  PLSTAT
0031  PLTFCNS
0032  RTN
;
; ---- add point (Zn, Hc)
0033  LBL 50
0034  RCL 96
0035  RCL 97
0036  Σ+
0037  RTN
;
; ---- HCZ; X = Zn
0038  LBL 52
0039  XEQ "HCZ"
0040  RCL 97
0041  RTN
0042  END
